# =============================================================================
# NEC IX2207 設定例: OCN MAP-E (DHCPv6-PD 動的IP / VLAN4)
#   機種: NEC IX2207 Software Version 10.8.46
#   構成: GigaEthernet0.0 = WAN (DHCPv6-PD / MAP-E), GigaEthernet1.0/2.0 = LAN
# =============================================================================

# --- 1. 基本・ロギング設定 ---
hostname ix-PD-dyn
timezone +09 00
terminal default-length 0
terminal default-width 512
!
logging buffered 8388608
logging subsystem all warn
logging subsystem dh6 debug
logging subsystem mape debug
logging timestamp datetime

# --- 2. NTP (時刻同期) 設定 ---
ntp ip enable
ntp ipv6 enable
ntp server dhcpv6

# --- 3. キャッシュ・デフォルトルート・DHCPv6 設定 ---
# ※ MAP-E トンネルを起動するため、ia-pd subscriber には ::/64 eui-64 の割り当てが必須
#   (WAN 側 GigaEthernet0.0 や Loopback0.0 には割り当て不可)

ip ufs-cache max-entries 20000
ip ufs-cache enable
ip route default Tunnel0.0
ip dhcp enable
!
ipv6 ufs-cache max-entries 10000
ipv6 ufs-cache enable
ipv6 dhcp enable
!
ipv6 dhcp client-profile dhcpv6-cl
  option-request dns-servers
  option-request domain-search-list
  option-request ntp-servers
  ia-pd subscriber GigaEthernet1.0 ::/64 eui-64
  ia-pd subscriber GigaEthernet2.0 0:0:0:ff::/64 eui-64
!
ipv6 dhcp server-profile dhcpv6-sv10
  dns-server dhcp
!
ipv6 dhcp server-profile dhcpv6-sv20
  dns-server dhcp

# --- 4. WAN (GigaEthernet0.0) 設定 ---
interface GigaEthernet0.0
  no ip address
  ipv6 enable
  ipv6 dhcp client dhcpv6-cl
  no shutdown

# --- 5. LAN (GigaEthernet1.0 / GigaEthernet2.0) 設定 ---
interface GigaEthernet1.0
  description // LAN10
  ip address 192.168.10.1/24
  ipv6 enable
  ipv6 dhcp server dhcpv6-sv10
  ipv6 nd ra enable
  ipv6 nd ra other-config-flag
  no shutdown
!
interface GigaEthernet2.0
  description // LAN20
  ip address 192.168.20.1/24
  ipv6 enable
  ipv6 dhcp server dhcpv6-sv20
  ipv6 nd ra enable
  ipv6 nd ra other-config-flag
  no shutdown

# --- 6. MAP-E トンネル設定 (Tunnel0.0) ---
interface Tunnel0.0
  tunnel mode map-e ocn
  ip address map-e
  ip tcp adjust-mss auto
  ip napt enable
  ip napt hairpinning
  ip napt static Tunnel0.0 1
  no shutdown

# --- 7. 未使用ポート・初期仮想インターフェース (参考) ---
device GigaEthernet0
!
device GigaEthernet1
!
device GigaEthernet2
!
device USB0
  shutdown
!
device USB1
  shutdown
!
interface USB-Serial0.0
  encapsulation ppp
  no auto-connect
  no ip address
  shutdown
!
interface USB-Serial1.0
  encapsulation ppp
  no auto-connect
  no ip address
  shutdown
!
interface Loopback0.0
  no ip address
!
interface Null0.0
  no ip address
!
interface Null1.0
  no ip address


# =============================================================================
# 運用メモ: 再接続手順
# =============================================================================
clear map-e status
