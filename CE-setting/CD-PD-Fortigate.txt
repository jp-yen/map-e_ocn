## 1. WAN インターフェース設定 (DHCPv6-PD /56 受信)

# 1-1. WAN1 で DHCPv6-PD (/56) を受信し、WAN1自身にもプロビジョニング用IPv6 (/64) を割り当てる
config system interface
    edit "wan1"
        set vdom "root"
        set allowaccess ping
        set type physical
        set role wan
        config ipv6
            set ip6-mode delegated
            set ip6-allowaccess ping
            set dhcp6-prefix-delegation enable
            config dhcp6-iapd-list
                edit 1
                    set prefix-hint ::/56
                next
            end
            # vned プロビジョニング用 (WAN自身にプレフィックスを認識させる)
            set ip6-delegated-prefix-iaid 1
            set ip6-upstream-interface "wan1"
            # 1つ目のサブネット
            set ip6-subnet ::1/64
        end
    next
end

# 1-2. (任意) 取得した /56 から Loopback へ別の /64 を切り出して割り当てる場合
# ※ wan1 で 1つ目のサブネット (::1/64) を使用しているため、Loopback は 2つ目以降を指定します
# LAN へ IPv6 を流す例です。ループバックに IPv6 が必要な訳ではありません。
config system interface
    edit "loopback-pd1"
        set vdom "root"
        set type loopback
        set role lan
        config ipv6
            set ip6-mode delegated
            set ip6-allowaccess ping
            set ip6-delegated-prefix-iaid 1
            set ip6-upstream-interface "wan1"
            # 2つ目のサブネット
            set ip6-subnet ::1:0:0:0:1/64
        end
    next
    edit "loopback-pd2"
        set vdom "root"
        set type loopback
        set role lan
        config ipv6
            set ip6-mode delegated
            set ip6-allowaccess ping
            set ip6-delegated-prefix-iaid 1
            set ip6-upstream-interface "wan1"
            # 3つ目のサブネット
            set ip6-subnet ::2:0:0:0:1/64
        end
    next
end


## 2. MAP-E プロビジョニング用 CA 証明書の登録 (検証環境用)

# サーバー側で生成された CA 証明書 (mape-provisioning-server/server.crt) をインポート
config vpn certificate ca
    edit "test_OCN_MAP_E_CA"
        set ca "-----BEGIN CERTIFICATE-----
ここに server.crt の内容を貼り付ける
-----END CERTIFICATE-----"
    next
end


## 3. VNE インターフェース (MAP-E 自動トンネル) 設定
# WAN1 を指定するだけで、モード(map-e)や各種パラメータはデフォルトで自動適用されます
config system vne-interface
    edit "vne.wan1"
        set interface "wan1"
    next
end


## 4. 静的ルーティング (IPv4 デフォルトルート)
config router static
    edit 1
        set dst 0.0.0.0 0.0.0.0
        set device "vne.wan1"
    next
end
